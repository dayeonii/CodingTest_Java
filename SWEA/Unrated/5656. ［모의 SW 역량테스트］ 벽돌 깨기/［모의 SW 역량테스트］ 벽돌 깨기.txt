import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

/*
 * [문제]
 * - N개의 벽돌을 떨어뜨려서 제거하기 (N번의 기회)
 * - 벽돌에 적힌 숫자-1 만큼 가로세로로 제거된다
 * 		- 벽돌에 3이라고 적혀있으면, 그 칸을 중심으로 상하좌우 2칸씩 더 제거
 * 		- 제거될 때 영향을 받는 다른 벽돌들도 같이 제거된다
 * - 제거하고 남은 벽돌의 개수 출력
 * 
 * [풀이]
 * - 완탐 : N번의 기회 -> 중복 순열 (같은 자리에 연속해서 떨어뜨릴 수 있음)
 * 		- N 최대 4, W 최대 12 -> W^N -> 12^4 -> 20736 (가능!)
 * - dfs로 중복 순열 구하기
 * - bfs로 연쇄 작용 벽돌 제거
 * - 백트래킹으로 원복
 * - 제거되고 빈 칸은 중력 작용하기
 * */

public class Solution {

	static int result;
	static int N, W, H;
	static int[][] map;

	static int[][] delta = { { -1, 0 }, { 1, 0 }, { 0, -1 }, { 0, 1 } };

	static class Point {
		int r, c;

		Point(int r, int c) {
			this.r = r;
			this.c = c;
		}
	}

	public static void main(String[] args) throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringBuilder sb = new StringBuilder();

		int T = Integer.parseInt(br.readLine());
		for (int tc = 1; tc <= T; tc++) {
			StringTokenizer st = new StringTokenizer(br.readLine());

			// 입력
			N = Integer.parseInt(st.nextToken());
			W = Integer.parseInt(st.nextToken());
			H = Integer.parseInt(st.nextToken());

			map = new int[H][W];
			for (int i = 0; i < H; i++) {
				st = new StringTokenizer(br.readLine());
				for (int j = 0; j < W; j++) {
					map[i][j] = Integer.parseInt(st.nextToken());
				}
			}

			// 초기화
			// 남은 블록에 가장 적을 때를 구해야 하니까, 최댓값으로 초기화
			result = Integer.MAX_VALUE;

			// 풀이
			// 블록을 떨어뜨리는 다양한 케이스를 살펴보면 된다 - 중복 순열을 호출하고 그 안에서 다양한 로직이 이루어짐
			dupPerm(0);

			// 결과 출력
			sb.append('#').append(tc).append(' ').append(result).append('\n');
		}
		System.out.print(sb);
	}

	private static void dupPerm(int count) {
		// 종료 조건
		// 벽돌을 N번 다 떨어뜨렸으면 - 제거되지 않은 남은 블럭의 개수를 세서
		// 전역 result 와 비교하여 더 작은 값으로 갱신하면 된다
		if (count == N) {
			int localResult = 0;

			for (int i = 0; i < H; i++) {
				for (int j = 0; j < W; j++) {
					if (map[i][j] != 0) {
						localResult++;
					}
				}
			}

			result = Math.min(result, localResult);
			return;
		}

		// 백트래킹 복원을 위해 현재 상태의 맵 저장
		int[][] backupMap = new int[H][W];
		for (int i = 0; i < H; i++) {
			for (int j = 0; j < W; j++) {
				backupMap[i][j] = map[i][j];
			}
		}

		// 재귀 호출 하기
		// 벽돌을 몇 번째 열에 놓을까? 를 고민하면 됨
		for (int j = 0; j < W; j++) {
			// j번째 줄에 놓기로 정했다면 - 그 줄에 있는 맨 위에 쌓여있는 블록이 꺠지는 거니까
			for (int i = 0; i < H; i++) {
				if (map[i][j] == 0) {
					continue; // 아래 칸으로 이동
				}

				// 블록이 있다면 얘를 깨고 - 연쇄 작용 재귀 호출
				Point first = new Point(i, j);
				blockRemove(first);
				break; // 연쇄 작용으로 다 깼으면, 지금 줄에 놓은 것에 대해서는 처리 끝
			}

			// 블록 제거 후 압축 연산 진행
			mapCompress();

			// 다음 카운트로 놓으러 ㄱㄱ
			dupPerm(count + 1);

			// 백트래킹 원복 - 재귀 호출이 끝나면
			for (int x = 0; x < H; x++) {
				for (int y = 0; y < W; y++) {
					map[x][y] = backupMap[x][y];
				}
			}
		}
	}

	private static void mapCompress() {

		// 세로줄을 순서대로 보면서 중간에 0이 있으면 밑으로 내려주면 된다
		for(int j=0; j<W; j++) {
			
			// 제일 바닥부터 올라오면서 탐색하다가
			// 0이 나오면 그 자리를 기록하고, 공중에 떠있는 애를 그 자리로 옮기면 됨
			for(int i=H-1; i>=0; i--) {
				
				// 지금 이 세로줄에 대해서 압축처리가 완료되었는가 - 다음 열으로 넘어가도 되는가
				boolean completeCompress = false;
				
				if(map[i][j]==0) {
					int idx = i;	// 공중에 떠있는 애를 찾을 인덱스
					
					while(true) {
						// 공중에 떠있는 애가 없다 (idx를 계속 위로 올리면서 봤는데 없어 -> idx가 맵 밖으로 나감)
						if(idx<0) {
							completeCompress = true;
							break;
						}
						
						// 공중에 떠있는 블록을 발견하면 (지금 idx는 공중에 떠있는 애)
						if(map[idx][j]!=0) {
							map[i][j] = map[idx][j];
							map[idx][j] = 0;
							break;	// while문을 탈출해서, 해당 세로줄의 그 다음 공중에 떠있는 애를 찾으러간다
						}
						
						// 공중에 떠있는 애 찾으러 가기
						idx--;
					}
				}
				
				// 다음 줄으로 넘어가도 된다
				if(completeCompress) {
					break;
				}
			}
		}
		
	}

	private static void blockRemove(Point cur) {

		// 몇 칸 제거해야 하는지, 적혀있는 숫자 확인
		int range = map[cur.r][cur.c];

		// 해당 칸 제거
		map[cur.r][cur.c] = 0;

		// 사방으로 깨야 하는거 확인하기
		for (int d = 0; d < 4; d++) {

			// range-1 만큼 옆으로 쭉 가주면 된다
			for (int i = 1; i < range; i++) {
				int nr = cur.r + delta[d][0] * i;	// 1칸 옆, 2칸 옆, ... N-1칸 옆
				int nc = cur.c + delta[d][1] * i;

				if (!canGo(nr, nc)) {
					break;
				}
				
				// 다음 블럭이 1보다 크면 - 그 안에서 또 재귀 호출 연쇄 작용
				if(map[nr][nc]>1) {
					Point next = new Point(nr,nc);
					blockRemove(next);
				}
				
				// 다음 블럭이 1이면 걔만 깨주고 끝임
				else {
					map[nr][nc]=0;
				}
				
			}

		}

	}

	private static boolean canGo(int nr, int nc) {
		return nr >= 0 && nr < H && nc >= 0 && nc < W;
	}

}
